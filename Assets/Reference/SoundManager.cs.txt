using UnityEngine;

public class SoundManager : MonoBehaviour
{
    public SoundEnifi BtnShoot;
    public SoundEnifi BtnSwap;
    public SoundEnifi SoundMove;
    public SoundEnifi DeadSound;
    public SoundEnifi LandingSound;
    public SoundEnifi PullSound;
    public SoundEnifi AttackSound;

    public static SoundManager Instance;

    private void Awake()
    {
        if (Instance == null)
            Instance = this;
        else if (Instance != this)
        {
            Destroy(gameObject);
            return;
        }
    }

    public static void PlayVFX(SoundType sound)
    {
        if (Instance == null) return;

        switch (sound)
        {
            case SoundType.Move:
                if (Instance.SoundMove == null) return;
                Instance.SoundMove.Play();
                break;

            case SoundType.TapShoot:
                if (Instance.BtnShoot == null) return;
                Instance.BtnShoot.Play();
                break;
            case SoundType.Swap:
                if (Instance.BtnSwap == null) return;
                Instance.BtnSwap.Play();
                break;
            case SoundType.Dead:
                if (Instance.DeadSound == null) return;
                Instance.DeadSound.Play();
                break;
            case SoundType.Landing:
                if (Instance.LandingSound == null) return;
                Instance.LandingSound.Play();
                break;
            case SoundType.Attack:
                if (Instance.AttackSound == null) return;
                Instance.AttackSound.Play();
                break;
            case SoundType.Pull:
                if (Instance.PullSound == null) return;
                Instance.PullSound.Play();
                break;
        }
    }

    public static void StopVFX(SoundType sound)
    {
        if (Instance == null) return;

        switch (sound)
        {
            case SoundType.Move:
                if (Instance.SoundMove == null) return;
                Instance.SoundMove.Stop();
                break;
        }
    }
}

public enum SoundType
{
    Move,
    TapShoot,
    Swap,
    Landing,
    Pull,
    Attack,
    Dead
}

public enum SoundTypeIcon
{
    Win,
    Fail
}
